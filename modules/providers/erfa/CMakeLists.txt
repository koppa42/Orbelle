find_package(erfa 2.0.1 EXACT CONFIG REQUIRED)
find_path(PICOSHA2_INCLUDE_DIR NAMES picosha2.h REQUIRED)

# Compile the explicitly pinned, hash-checked snapshot into the provider. The installed
# SDK/wheel never needs the source checkout, network, or a mutable global ERFA table.
set(manifest "${PROJECT_SOURCE_DIR}/data/manifests/iers_leap_seconds_2026_07.json")
file(READ "${manifest}" manifest_json)
foreach(field snapshot_id source_url sha256 file utc_valid_from_mjd utc_valid_until_mjd)
    string(JSON ${field} GET "${manifest_json}" ${field})
endforeach()
set(data_file "${PROJECT_SOURCE_DIR}/${file}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manifest}" "${data_file}")
file(SHA256 "${data_file}" actual_hash)
if(NOT actual_hash STREQUAL sha256)
    message(FATAL_ERROR "IERS leap-second snapshot SHA256 mismatch: ${data_file}")
endif()
file(STRINGS "${data_file}" data_rows REGEX "^[ ]+[0-9]")
set(ORBITFORGE_LEAP_ROWS "")
foreach(row IN LISTS data_rows)
    if(NOT row MATCHES "^[ ]+([0-9]+)\\.0[ ]+[0-9]+[ ]+[0-9]+[ ]+[0-9]+[ ]+([0-9]+)[ ]*$")
        message(FATAL_ERROR "Invalid IERS leap-second record: ${row}")
    endif()
    string(APPEND ORBITFORGE_LEAP_ROWS "    {${CMAKE_MATCH_1}, ${CMAKE_MATCH_2}},\n")
endforeach()
configure_file(src/leap_seconds.hpp.in generated/leap_seconds.hpp @ONLY)

orbitforge_module(orbitforge_erfa Erfa src/provider.cpp)
target_include_directories(orbitforge_erfa PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(orbitforge_erfa SYSTEM PRIVATE "${PICOSHA2_INCLUDE_DIR}")
target_link_libraries(orbitforge_erfa PUBLIC OrbitForge::Core PRIVATE erfa::erfa)
foreach(component SDK Python)
    if(component STREQUAL "SDK")
        set(notice_dir "${CMAKE_INSTALL_DATADIR}/OrbitForge/notices")
    else()
        set(notice_dir "orbitforge/notices")
    endif()
    install(FILES "${ERFA_LICENSE_FILE}" DESTINATION "${notice_dir}" RENAME ERFA.txt COMPONENT ${component})
    install(FILES "${manifest}" "${data_file}" DESTINATION "${notice_dir}" COMPONENT ${component})
endforeach()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
