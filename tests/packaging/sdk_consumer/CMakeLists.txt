cmake_minimum_required(VERSION 3.28)
project(OrbelleSdkConsumer LANGUAGES CXX)
option(ORBELLE_EXPECT_TIME_FRAMES "Expect the installed SDK to include ERFA" OFF)
option(ORBELLE_EXPECT_SPICE "Expect the installed SDK to include CSPICE" OFF)
find_package(Orbelle CONFIG REQUIRED COMPONENTS Runtime)
add_executable(sdk_consumer main.cpp)
target_link_libraries(sdk_consumer PRIVATE Orbelle::Runtime)
target_compile_definitions(sdk_consumer PRIVATE
    ORBELLE_EXPECT_TIME_FRAMES=$<BOOL:${ORBELLE_EXPECT_TIME_FRAMES}>)
target_compile_definitions(sdk_consumer PRIVATE ORBELLE_EXPECT_SPICE=$<BOOL:${ORBELLE_EXPECT_SPICE}>)
enable_testing()
set(ORBELLE_EOP_MANIFEST "" CACHE FILEPATH "EOP manifest in the installed SDK")
if(ORBELLE_EXPECT_TIME_FRAMES AND NOT EXISTS "${ORBELLE_EOP_MANIFEST}")
    message(FATAL_ERROR "Provide ORBELLE_EOP_MANIFEST from the installed SDK")
endif()
set(ORBELLE_SPICE_MANIFEST "" CACHE FILEPATH "SPICE manifest in the installed SDK")
if(ORBELLE_EXPECT_SPICE AND NOT EXISTS "${ORBELLE_SPICE_MANIFEST}")
    message(FATAL_ERROR "Provide ORBELLE_SPICE_MANIFEST from the installed SDK")
endif()
add_test(NAME installed_sdk COMMAND sdk_consumer "${ORBELLE_EOP_MANIFEST}"
    "${ORBELLE_SPICE_MANIFEST}" "${CMAKE_CURRENT_BINARY_DIR}/spice-workspace")

add_executable(sdk_project_api project_api.cpp)
target_link_libraries(sdk_project_api PRIVATE Orbelle::Runtime)
add_test(NAME installed_project_api COMMAND sdk_project_api "${CMAKE_CURRENT_BINARY_DIR}/project-api.orbelle")
